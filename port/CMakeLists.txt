cmake_minimum_required(VERSION 3.20)

# ============================================================================
# Buffy the Vampire Slayer (Xbox, 2002) - static recompilation for Windows
#
#   buffy_tvs   the recompiled game (src/, src/recomp/gen/)
# ============================================================================

project(buffy_tvs C CXX)

set(BTVS_VERSION "0.0.1")

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Release builds keep full debug info (a .pdb beside the exe), so crash and
# hang reports name functions.
if(MSVC)
    add_compile_options($<$<CONFIG:Release>:/Zi>)
    add_link_options($<$<CONFIG:Release>:/DEBUG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
endif()

add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS BTVS_VERSION="${BTVS_VERSION}")

set(XBOXRECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../xboxrecomp" CACHE PATH
    "Path to the xboxrecomp toolkit root directory")
# A git worktree has the toolkit's submodules (glslang, Vulkan headers, volk,
# VMA) empty: use the main checkout's copies.
set(BTVS_MAIN "" CACHE PATH "The main checkout, when building in a git worktree")
if(BTVS_MAIN AND NOT EXISTS "${XBOXRECOMP_DIR}/third_party/glslang/CMakeLists.txt"
        AND EXISTS "${BTVS_MAIN}/xboxrecomp/third_party/glslang/CMakeLists.txt")
    set(XBOXRECOMP_THIRD_PARTY "${BTVS_MAIN}/xboxrecomp/third_party")
endif()
add_subdirectory(${XBOXRECOMP_DIR} ${CMAKE_BINARY_DIR}/xboxrecomp)

# -- The translated game code: btvs_recomp.lib ---------------------------------
# Generated by tools/regen.sh into the MAIN checkout's port/src/recomp/gen (it
# is not in git). A git worktree has no copy, so it uses the main checkout's
# generated code and its already-built library (BTVS_MAIN, set by build.sh):
# a worktree build then compiles only the port's own sources.
set(BTVS_GEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen")
set(BTVS_PREBUILT "")
if(BTVS_MAIN AND NOT EXISTS "${BTVS_GEN_DIR}/recomp_funcs.h")
    set(BTVS_GEN_DIR "${BTVS_MAIN}/port/src/recomp/gen")
    set(BTVS_PREBUILT "${BTVS_MAIN}/port/build/Release/btvs_recomp.lib")
endif()

if(BTVS_PREBUILT AND NOT EXISTS "${BTVS_PREBUILT}")
    message(FATAL_ERROR "No ${BTVS_PREBUILT}: build the main checkout first (bash port/tools/build.sh there).")
endif()
if(BTVS_PREBUILT)
    add_library(btvs_recomp STATIC IMPORTED)
    set_target_properties(btvs_recomp PROPERTIES IMPORTED_LOCATION "${BTVS_PREBUILT}")
else()
    file(GLOB RECOMP_GEN_SOURCES CONFIGURE_DEPENDS ${BTVS_GEN_DIR}/*.c)
    add_library(btvs_recomp STATIC ${RECOMP_GEN_SOURCES})
    target_include_directories(btvs_recomp PRIVATE
        ${BTVS_GEN_DIR} ${BTVS_GEN_DIR}/..
        ${CMAKE_CURRENT_SOURCE_DIR}/src ${XBOXRECOMP_DIR}/include)
    if(MSVC)
        target_compile_options(btvs_recomp PRIVATE /bigobj)
    endif()
endif()

set(GAME_SOURCES
    src/main.c
    src/recomp_manual.c
    src/btvs_settings.c
    src/btvs_menu.c
    src/btvs_xbecheck.c
)
# Graphics area. Its files hook functions named in src/manual_functions.json,
# so they join the build once the generated code has those hooks (after the
# regen that follows a change to that file); before it, the game builds
# without them.
set(BTVS_GRAPHICS OFF)
if(EXISTS "${BTVS_GEN_DIR}/recomp_funcs.h")
    file(STRINGS "${BTVS_GEN_DIR}/recomp_funcs.h" _btvs_gfx_hooks REGEX "D3DDevice_Swap_00595DB0_orig|XGetVideoFlags_00579D81_orig")
    list(LENGTH _btvs_gfx_hooks _btvs_gfx_n)
    if(_btvs_gfx_n GREATER 1)
        set(BTVS_GRAPHICS ON)
    endif()
endif()
if(BTVS_GRAPHICS)
    list(APPEND GAME_SOURCES
        src/btvs_boot.c
        src/btvs_native.c
        src/btvs_native_gpuwait.c
        src/btvs_native_shot.c
        src/btvs_native_game.c
        src/btvs_window.c
        src/btvs_frame.c
    )
endif()
message(STATUS "Graphics area in the build: ${BTVS_GRAPHICS}")

add_executable(${PROJECT_NAME} WIN32 ${GAME_SOURCES})

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${BTVS_GEN_DIR} ${BTVS_GEN_DIR}/..
    ${XBOXRECOMP_DIR}/include
    ${XBOXRECOMP_DIR}/src/gpu
)

if(BTVS_GRAPHICS)
    target_compile_definitions(${PROJECT_NAME} PRIVATE BTVS_GRAPHICS=1)
endif()

target_link_libraries(${PROJECT_NAME} PRIVATE
    btvs_recomp
    xboxrecomp
    d3d11 dxgi dxguid xinput winmm dbghelp
)

# -- Controls (docs/controls.md) -----------------------------------------------
set(BTVS_INPUT_SOURCES
    src/btvs_input.c
    src/btvs_input_xinput.c
    src/btvs_input_sdl.c
    src/btvs_input_kbm.c
)
target_sources(${PROJECT_NAME} PRIVATE ${BTVS_INPUT_SOURCES})

# The XAPI replacements define functions the recompiler otherwise generates, so
# they are only compiled in once the generated code was made with them listed
# in src/manual_functions.json (tools/regen.sh leaves a copy of the list it
# used beside the generated code). Other areas can test BTVS_MANUAL_APPLIED too.
set(BTVS_MANUAL_APPLIED "")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${BTVS_GEN_DIR}/manual_functions.applied.json")
if(EXISTS "${BTVS_GEN_DIR}/manual_functions.applied.json")
    file(READ "${BTVS_GEN_DIR}/manual_functions.applied.json" BTVS_MANUAL_APPLIED)
endif()
if(BTVS_MANUAL_APPLIED MATCHES "XInputGetState_006105EF")
    target_sources(${PROJECT_NAME} PRIVATE src/btvs_input_xapi.c)
else()
    message(WARNING "Controls: the generated code still has the game's own XInput functions; "
                    "run tools/regen.sh in the main checkout to switch to the PC input layer")
endif()

# Compile-only check of the XAPI replacements, for before that regen:
#   bash tools/build.sh btvs_input_xapi_check
add_library(btvs_input_xapi_check OBJECT EXCLUDE_FROM_ALL src/btvs_input_xapi.c)
target_include_directories(btvs_input_xapi_check PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src ${BTVS_GEN_DIR} ${BTVS_GEN_DIR}/.. ${XBOXRECOMP_DIR}/include)

# The input layer on its own: self-test, pad-script player, live pad printer.
add_executable(btvs_input_test src/btvs_input_test.c ${BTVS_INPUT_SOURCES} src/btvs_settings.c)
target_include_directories(btvs_input_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)

# SDL3.dll (DualSense, Switch and other non-Xbox pads) is loaded at run time and
# is optional; the build fetches the official release and puts it beside the exe.
option(BTVS_FETCH_SDL "Download SDL3.dll for the SDL controller backend" ON)
set(BTVS_SDL_VERSION "3.4.20")
set(BTVS_SDL_SHA256 "368bed8b24e44ad9ea372990814f2343fcfb9e468abb7f8f4a3bc146c7304752")
set(BTVS_SDL_DIR "${CMAKE_BINARY_DIR}/sdl3")
if(BTVS_FETCH_SDL AND NOT EXISTS "${BTVS_SDL_DIR}/SDL3.dll")
    set(_sdl_zip "${CMAKE_BINARY_DIR}/SDL3-${BTVS_SDL_VERSION}-win32-x64.zip")
    file(DOWNLOAD
        "https://github.com/libsdl-org/SDL/releases/download/release-${BTVS_SDL_VERSION}/SDL3-${BTVS_SDL_VERSION}-win32-x64.zip"
        "${_sdl_zip}" EXPECTED_HASH SHA256=${BTVS_SDL_SHA256} TLS_VERIFY ON STATUS _sdl_status)
    list(GET _sdl_status 0 _sdl_code)
    if(_sdl_code EQUAL 0)
        file(ARCHIVE_EXTRACT INPUT "${_sdl_zip}" DESTINATION "${BTVS_SDL_DIR}")
    else()
        message(WARNING "Controls: could not download SDL3 (${_sdl_status}); the game still builds and runs, without the SDL backend")
        file(REMOVE "${_sdl_zip}")
    endif()
endif()
if(EXISTS "${BTVS_SDL_DIR}/SDL3.dll")
    foreach(_t ${PROJECT_NAME} btvs_input_test)
        add_custom_command(TARGET ${_t} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${BTVS_SDL_DIR}/SDL3.dll" "$<TARGET_FILE_DIR:${_t}>")
    endforeach()

# -- "Buffy Launcher.exe": install from the disc image, then Play (launcher/) --
add_executable(buffy_tvs_launcher WIN32 launcher/btvs_launcher.c)
set_target_properties(buffy_tvs_launcher PROPERTIES OUTPUT_NAME "Buffy Launcher")
target_compile_definitions(buffy_tvs_launcher PRIVATE UNICODE _UNICODE)
target_link_libraries(buffy_tvs_launcher PRIVATE comdlg32 shell32 ole32 uuid dwmapi bcrypt)
if(MSVC)
    target_link_options(buffy_tvs_launcher PRIVATE /ENTRY:wWinMainCRTStartup)
endif()
