cmake_minimum_required(VERSION 3.20)

# ============================================================================
# Buffy the Vampire Slayer (Xbox, 2002) - static recompilation for Windows
#
#   buffy_tvs   the recompiled game (src/, src/recomp/gen/)
# ============================================================================

project(buffy_tvs C CXX)

set(BTVS_VERSION "0.0.1")

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Release builds keep full debug info (a .pdb beside the exe), so crash and
# hang reports name functions.
if(MSVC)
    add_compile_options($<$<CONFIG:Release>:/Zi>)
    add_link_options($<$<CONFIG:Release>:/DEBUG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
endif()

add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS BTVS_VERSION="${BTVS_VERSION}")

set(XBOXRECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../xboxrecomp" CACHE PATH
    "Path to the xboxrecomp toolkit root directory")
add_subdirectory(${XBOXRECOMP_DIR} ${CMAKE_BINARY_DIR}/xboxrecomp)

# -- The translated game code: btvs_recomp.lib ---------------------------------
# Generated by tools/regen.sh into the MAIN checkout's port/src/recomp/gen (it
# is not in git). A git worktree has no copy, so it uses the main checkout's
# generated code and its already-built library (BTVS_MAIN, set by build.sh):
# a worktree build then compiles only the port's own sources.
set(BTVS_MAIN "" CACHE PATH "The main checkout, when building in a git worktree")
set(BTVS_GEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen")
set(BTVS_PREBUILT "")
if(BTVS_MAIN AND NOT EXISTS "${BTVS_GEN_DIR}/recomp_funcs.h")
    set(BTVS_GEN_DIR "${BTVS_MAIN}/port/src/recomp/gen")
    set(BTVS_PREBUILT "${BTVS_MAIN}/port/build/Release/btvs_recomp.lib")
endif()

if(BTVS_PREBUILT AND NOT EXISTS "${BTVS_PREBUILT}")
    message(FATAL_ERROR "No ${BTVS_PREBUILT}: build the main checkout first (bash port/tools/build.sh there).")
endif()
if(BTVS_PREBUILT)
    add_library(btvs_recomp STATIC IMPORTED)
    set_target_properties(btvs_recomp PROPERTIES IMPORTED_LOCATION "${BTVS_PREBUILT}")
else()
    file(GLOB RECOMP_GEN_SOURCES CONFIGURE_DEPENDS ${BTVS_GEN_DIR}/*.c)
    add_library(btvs_recomp STATIC ${RECOMP_GEN_SOURCES})
    target_include_directories(btvs_recomp PRIVATE
        ${BTVS_GEN_DIR} ${BTVS_GEN_DIR}/..
        ${CMAKE_CURRENT_SOURCE_DIR}/src ${XBOXRECOMP_DIR}/include)
    if(MSVC)
        target_compile_options(btvs_recomp PRIVATE /bigobj)
    endif()
endif()

set(GAME_SOURCES
    src/main.c
    src/recomp_manual.c
    src/btvs_settings.c
    src/btvs_menu.c
)
# Graphics area. Its files hook functions named in src/manual_functions.json,
# so they join the build once the generated code has those hooks (after the
# regen that follows a change to that file); before it, the game builds
# without them.
set(BTVS_GRAPHICS OFF)
if(EXISTS "${BTVS_GEN_DIR}/recomp_funcs.h")
    file(STRINGS "${BTVS_GEN_DIR}/recomp_funcs.h" _btvs_gfx_hooks REGEX "D3DDevice_Swap_00595DB0_orig|XGetVideoFlags_00579D81_orig")
    list(LENGTH _btvs_gfx_hooks _btvs_gfx_n)
    if(_btvs_gfx_n GREATER 1)
        set(BTVS_GRAPHICS ON)
    endif()
endif()
if(BTVS_GRAPHICS)
    list(APPEND GAME_SOURCES
        src/btvs_native.c
        src/btvs_native_gpuwait.c
        src/btvs_native_shot.c
        src/btvs_native_game.c
        src/btvs_window.c
        src/btvs_frame.c
    )
endif()
message(STATUS "Graphics area in the build: ${BTVS_GRAPHICS}")

add_executable(${PROJECT_NAME} WIN32 ${GAME_SOURCES})

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${BTVS_GEN_DIR} ${BTVS_GEN_DIR}/..
    ${XBOXRECOMP_DIR}/include
    ${XBOXRECOMP_DIR}/src/gpu
)

if(BTVS_GRAPHICS)
    target_compile_definitions(${PROJECT_NAME} PRIVATE BTVS_GRAPHICS=1)
endif()

target_link_libraries(${PROJECT_NAME} PRIVATE
    btvs_recomp
    xboxrecomp
    d3d11 dxgi dxguid xinput winmm dbghelp
)
