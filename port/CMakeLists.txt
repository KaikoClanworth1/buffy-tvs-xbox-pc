cmake_minimum_required(VERSION 3.20)

# ============================================================================
# Buffy the Vampire Slayer (Xbox, 2002) - static recompilation for Windows
#
#   buffy_tvs   the recompiled game (src/, src/recomp/gen/)
# ============================================================================

project(buffy_tvs C CXX)

set(BTVS_VERSION "0.0.1")

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Release builds keep full debug info (a .pdb beside the exe), so crash and
# hang reports name functions.
if(MSVC)
    add_compile_options($<$<CONFIG:Release>:/Zi>)
    add_link_options($<$<CONFIG:Release>:/DEBUG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
endif()

add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS BTVS_VERSION="${BTVS_VERSION}")

set(XBOXRECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../xboxrecomp" CACHE PATH
    "Path to the xboxrecomp toolkit root directory")
add_subdirectory(${XBOXRECOMP_DIR} ${CMAKE_BINARY_DIR}/xboxrecomp)

# Generated by tools/regen.sh; the file count follows --split, so they are globbed.
file(GLOB RECOMP_GEN_SOURCES CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen/*.c
)

set(GAME_SOURCES
    src/main.c
    src/recomp_manual.c
)

add_executable(${PROJECT_NAME} WIN32 ${GAME_SOURCES} ${RECOMP_GEN_SOURCES})

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/src/recomp
    ${XBOXRECOMP_DIR}/include
)

if(MSVC)
    target_compile_options(${PROJECT_NAME} PRIVATE /bigobj)
endif()

target_link_libraries(${PROJECT_NAME} PRIVATE
    xboxrecomp
    d3d11 dxgi dxguid xinput winmm dbghelp
)
